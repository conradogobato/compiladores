.data
newline: .asciiz "\n"
.text
.globl main


gcd:
    # --- Prologo ---
    addiu $sp, $sp, -16   # Aloca 16 bytes na pilha
    sw $ra, 12($sp)       # Guarda o endereco de retorno
    sw $fp, 8($sp)       # Guarda o frame pointer antigo
    move $fp, $sp         # Estabelece o novo frame pointer
    # Salvando parametros na pilha
    sw $a0, -4($fp)       # Salva parametro 'u'
    sw $a1, -8($fp)       # Salva parametro 'v'

    lw $t0, -8($fp)     # Carrega 'v' para $t0
    li $t1, 0            # Carrega imediato 0 para $t1
    seq $t2, $t0, $t1      # $t2 = ($t0 == $t1)
    beqz $t2, L0         # Salta para L0 se $t2 for falso (0)
    lw $t3, -4($fp)     # Carrega 'u' para $t3
    move $v0, $t3         # Prepara valor de retorno de um registrador
    j L_end_gcd               # Pula para o epilogo da funcao
    j L1
L0:
    lw $t4, -8($fp)     # Carrega 'v' para $t4
    move $a0, $t4        # Prepara parametro 1
    lw $t5, -4($fp)     # Carrega 'u' para $t5
    lw $t6, -8($fp)     # Carrega 'v' para $t6
    div $t5, $t6
    mflo $t7              # Quociente da divisao para $t7
    move $a1, $t7        # Prepara parametro 2
    jal gcd
    move $t8, $v0         # Captura valor de retorno em $t8
    move $v0, $t8         # Prepara valor de retorno de um registrador
    j L_end_gcd               # Pula para o epilogo da funcao
L1:
L_end_gcd:
    # --- Epilogo ---
    move $sp, $fp         # Restaura o ponteiro de pilha para a base do frame
    lw $ra, 12($sp)       # Restaura o endereco de retorno
    lw $fp, 8($sp)       # Restaura o frame pointer antigo
    addiu $sp, $sp, 16    # Liberta o espaco da pilha
    jr $ra                # Retorna para a funcao que chamou


main:
    # --- Prologo ---
    addiu $sp, $sp, -16   # Aloca 16 bytes na pilha
    sw $ra, 12($sp)       # Guarda o endereco de retorno
    sw $fp, 8($sp)       # Guarda o frame pointer antigo
    move $fp, $sp         # Estabelece o novo frame pointer

    li $t9, 12            # Carrega imediato 12 para $t9
    sw $t9, -4($fp)     # Armazena $t9 em 'x'
    li $t10, 8            # Carrega imediato 8 para $t10
    sw $t10, -8($fp)     # Armazena $t10 em 'y'
    lw $t11, -4($fp)     # Carrega 'x' para $t11
    move $a0, $t11        # Prepara parametro 1
    lw $t12, -8($fp)     # Carrega 'y' para $t12
    move $a1, $t12        # Prepara parametro 2
    jal gcd
    move $t13, $v0         # Captura valor de retorno em $t13
    move $a0, $t13        # Prepara parametro 1
    # Chamada para output via syscall
    move $a0, $t13         # Move valor para argumento de print
    li $v0, 1             # Syscall para imprimir inteiro
    syscall
    la $a0, newline       # Carrega endereço de nova linha
    li $v0, 4             # Syscall para imprimir string
    syscall
L_end_main:
    # --- Epilogo ---
    move $sp, $fp         # Restaura o ponteiro de pilha para a base do frame
    lw $ra, 12($sp)       # Restaura o endereco de retorno
    lw $fp, 8($sp)       # Restaura o frame pointer antigo
    addiu $sp, $sp, 16    # Liberta o espaco da pilha
    j exit_program


# Finalizacao padrao do programa
exit_program:
    # No epilogo de main, pulamos para ca
    li $v0, 10          # Syscall para sair
    syscall
